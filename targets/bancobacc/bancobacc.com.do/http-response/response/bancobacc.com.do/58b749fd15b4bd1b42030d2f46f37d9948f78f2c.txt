(function ($) {
    var map = new GMaps({
        el: '#map-container',
        lat: 18.485169941932313,
        lng: -69.94005382061005,
        zoom: 16
    });
    var unhiglightMarkers = function () {
        for (var i = 0, m; m = map.markers[i]; i++) {
            if (m.infoWindow) {
                m.windowIsOpen = false;
                m.setIcon(m.defaultIcon);
            }
        }
    };
    var i = 0;

    $('.map-marker').each(function () {
        
        var marker = map.createMarker($(this).data('marker'));

        google.maps.event.addListener(marker, 'click', function() {
            var markerLatitude = marker.getPosition().lat();
            var markerLongitude = marker.getPosition().lng();

            window.open(`https://maps.google.com/maps?q=loc:${markerLatitude},${markerLongitude}`, '_blank');

        });
        marker.windowIsOpen = false;
        marker.addListener('mouseover', function () {
            marker.setIcon(marker.selectedIcon);
        });
        marker.addListener('mouseout', function () {
            if (!marker.windowIsOpen) {
                marker.setIcon(marker.defaultIcon);
            }
        });
        if (marker.infoWindow) {
            marker.infoWindow.addListener('domready', function () {
                unhiglightMarkers();
                marker.windowIsOpen = true;
                marker.setIcon(marker.selectedIcon);
            });
            marker.infoWindow.addListener('closeclick', function () {
                unhiglightMarkers();
                marker.windowIsOpen = false;
            });
        }
        $(this).data('markerObject', marker);
        map.addMarker(marker);
        if(i == 0) {
            map.panTo(marker.position);
        }
        i++;
    }).click(function (e) {
        // set active, center and show infowindow
//          marker.setIcon(marker.selectedIcon);
        var marker = $(this).data('markerObject');
        map.hideInfoWindows();
        marker.infoWindow.open(map, marker);
        map.setZoom(16);
        map.panTo(marker.position);
    }).hover(function () {
        var marker = $(this).data('markerObject');
        marker.setIcon(marker.selectedIcon);
    }, function () {
        var marker = $(this).data('markerObject');
        if (!marker.windowIsOpen) {
            marker.setIcon(marker.defaultIcon);
        }
    });
})(jQuery);
