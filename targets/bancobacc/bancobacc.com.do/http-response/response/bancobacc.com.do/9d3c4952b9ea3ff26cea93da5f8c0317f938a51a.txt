(function($){
    window.InfiniteScroll = function (options) {
        var opts = $.extend({
            selector: null,
            container: null,
            maxPage: Infinity,
            page: 1,
            loadCallback: function(page){
                console.log('TODO: IMPLEMENT loadCallback')
            },
            contentCallback: function(container, items){
                console.log('TODO: IMPLEMENT contentCallback')
            },
            loadingCallback : function(show) {
                if(show){
                    $(opts.selector).addClass('loading');
                }else{
                    $(opts.selector).removeClass('loading');
                }
            },
            offset: 150
        }, options);
        var doingCall = false;
        var unsetListeners = function(){
            $(window).off('scroll', scrollListener);
            $(opts.selector).off('click', clickListener);
            $(opts.selector).fadeOut();
        };
        var loadContent = function(){
            if(doingCall){
                return;
            }
            doingCall = true;
            opts.loadingCallback(true);
            opts.loadCallback(opts.page).then(function(items){
                opts.contentCallback(opts.container,items);
                opts.page++;
                if(opts.page > opts.maxPage){
                    unsetListeners();
                }
            }, unsetListeners).always(function() {
                opts.loadingCallback(false);
                doingCall = false;
            });
        };
        var scrollListener = function () {
            if  ($(window).scrollTop() >= $(document).height() - ($(window).height() + opts.offset)){
                loadContent();
            }
        };
        var clickListener = function (e) {
            e.preventDefault();
            loadContent();
        };
        // Load first time content, bind infinite scroll and bind next click event
        $(opts.selector).one('click', function (e) {
            e.preventDefault();
            loadContent();
            $(window).scroll(scrollListener);
            $(opts.selector).on('click', clickListener);
        });
    };
})(jQuery, window);
