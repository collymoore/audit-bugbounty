(function($){
    var search, sort,perPage, currentPage, dealerTemplate, maxPage, $container = $('#dealers-container');
    search = $container.data('ds');
    sort = $container.data('dsort');
    currentPage = $container.data('dpage');
    maxPage = $container.data('maxPage');
    dealerTemplate = Handlebars.compile($("#dealer-card-template").html());

    var loadDealers = function (page) {
        var deferred = $.Deferred();
        $.get(bacc.admin_url, {
            'action' : 'get_dealers',
            'ds': search,
            'dsort': sort,
            'page' : page
        }, function(dealers){
            if(dealers.length){
                deferred.resolve(dealers);
            }else{
                deferred.reject();
            }
        },'json');
        return deferred.promise();
    };
    var insertDealers = function(container, dealers){
        $.each(dealers,function(index, dealer){
            var content = $(dealerTemplate(dealer));
            content.hide();
            container.append(content);
            content.fadeIn();
        });
        loadDealersMap();
    };
    InfiniteScroll({
        selector: '.infinite-scroll-btn',
        container: $container,
        page: currentPage,
        maxPage: maxPage,
        loadCallback: loadDealers,
        contentCallback: insertDealers
    });

    var loadDealersMap = function(){
        $('.dealer-map:not(.loaded)').each(function(index, dealerMap){
            var $dealerMap = $(dealerMap);
            $dealerMap.css({height: '230px', width: '100%'}).addClass('loaded');
            var map = new GMaps({
                el: dealerMap,
                lat: $dealerMap.data('lat'),
                lng: $dealerMap.data('lng'),
                zoom: 14,
                draggable: true,
                scrollwheel: true

            });
            map.addMarker({
                'lat': $dealerMap.data('lat'),
                'lng': $dealerMap.data('lng'),
                'icon': {
                    'url': bacc.theme_url + '/assets/images/dealer-marker.png'
                }
            });
        });
    };
    loadDealersMap();

})(jQuery);
